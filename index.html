About My Project

The Job Portal Platform SQL Project is a database management project developed using SQL. It manages information about candidates, companies, recruiters, jobs, applications, interviews, and skills.

This project demonstrates database creation, table design, primary keys, foreign keys, data insertion, filtering, sorting, aggregate functions, joins, and views.

The project also includes a portfolio website developed using HTML and CSS to present the database structure and SQL queries.

Database Relationships

  • Candidates → Applications: One candidate can submit multiple applications.
  • Jobs → Applications: One job can receive multiple applications.
  • Companies → Jobs: One company can post multiple jobs.
  • Recruiters → Jobs: One recruiter can be associated with multiple jobs.
  • Applications → Interviews: One application can have multiple interview records.
  • Jobs → Interviews: One job can be associated with multiple interview records.
  • Candidates → Candidate_Skills: One candidate can have multiple skill records.
  • Skills → Candidate_Skills: One skill can be associated with multiple candidates.

Note: The Recruiters table contains a company_id column, but the supplied SQL definition does not declare it as a foreign key. The Jobs table does declare foreign keys to Companies and Recruiters.

02. Create Tables 
CREATE TABLE Candidates (
    candidate_id INT PRIMARY KEY,
    full_name VARCHAR(50),
    email VARCHAR(50),
    phone VARCHAR(20),
    location VARCHAR(50),
    experience_years INT,
    resume_url VARCHAR(50),
    created_at DATE
);

CREATE TABLE Companies (
    company_id INT PRIMARY KEY,
    company_name VARCHAR(50),
    industry VARCHAR(50),
    website VARCHAR(50),
    location VARCHAR(50),
    about VARCHAR(100),
    created_at DATE
);

CREATE TABLE Recruiters (
    recruiter_id INT PRIMARY KEY,
    company_id INT,
    full_name VARCHAR(100),
    email VARCHAR(100),
    phone VARCHAR(15),
    designation VARCHAR(50),
    created_at DATE
);

CREATE TABLE Jobs (
    job_id INT PRIMARY KEY,
    company_id INT,
    recruiter_id INT,
    job_title VARCHAR(100),
    job_description VARCHAR(255),
    location VARCHAR(100),
    employment_type VARCHAR(50),
    experience_required INT,
    salary_min DECIMAL(10,2),
    salary_max DECIMAL(10,2),
    status VARCHAR(20),
    created_at DATE,
    FOREIGN KEY (company_id)
        REFERENCES Companies(company_id),
    FOREIGN KEY (recruiter_id)
        REFERENCES Recruiters(recruiter_id)
);

CREATE TABLE Applications (
    application_id INT PRIMARY KEY,
    candidate_id INT,
    job_id INT,
    applied_at DATE,
    status VARCHAR(30),
    cover_letter VARCHAR(255),
    FOREIGN KEY (candidate_id)
        REFERENCES Candidates(candidate_id),
    FOREIGN KEY (job_id)
        REFERENCES Jobs(job_id)
);

CREATE TABLE Interviews (
    interview_id INT PRIMARY KEY,
    application_id INT,
    job_id INT,
    interview_round VARCHAR(50),
    scheduled_at DATE,
    interview_type VARCHAR(20),
    status VARCHAR(20),
    feedback VARCHAR(255),
    created_at DATE,
    FOREIGN KEY (application_id)
        REFERENCES Applications(application_id),
    FOREIGN KEY (job_id)
        REFERENCES Jobs(job_id)
);

CREATE TABLE Skills (
    skill_id INT PRIMARY KEY,
    skill_name VARCHAR(50),
    created_at DATE
);

CREATE TABLE Candidate_Skills (
    candidate_skill_id INT PRIMARY KEY,
    candidate_id INT,
    skill_id INT,
    proficiency_level VARCHAR(20),
    years_of_experience INT,
    FOREIGN KEY (candidate_id)
        REFERENCES Candidates(candidate_id),
    FOREIGN KEY (skill_id)
        REFERENCES Skills(skill_id)
);

05. ORDER BY, GROUP BY and HAVING 
SELECT *
FROM Jobs
ORDER BY salary_min;

SELECT
    company_id,
    COUNT(*) AS total_jobs
FROM Jobs
GROUP BY company_id;

SELECT
    company_id,
    COUNT(*) AS total_jobs
FROM Jobs
GROUP BY company_id
HAVING COUNT(*) >= 1;
06. Aggregate Functions 
SELECT COUNT(*) AS total_candidates
FROM Candidates;

SELECT MAX(salary_max) AS highest_salary
FROM Jobs;

SELECT MIN(salary_min) AS lowest_salary
FROM Jobs;

SELECT SUM(salary_max) AS total_max_salary
FROM Jobs;

SELECT AVG(salary_max) AS average_max_salary
FROM Jobs;
07. INNER JOIN 
SELECT
    Companies.company_name,
    Recruiters.full_name,
    Recruiters.designation
FROM Companies
INNER JOIN Recruiters
    ON Companies.company_id = Recruiters.company_id;
08. LIMIT and OFFSET 
SELECT *
FROM Candidates
LIMIT 3 OFFSET 2;

SELECT *
FROM Candidates
LIMIT 2, 3;

SELECT
    job_title,
    salary_max
FROM Jobs
ORDER BY salary_max DESC
LIMIT 2;

SELECT
    job_title,
    salary_max
FROM Jobs
ORDER BY salary_max DESC
LIMIT 2 OFFSET 2;
09. Database Views 
CREATE VIEW Candidate_Public_View AS
SELECT
    candidate_id,
    full_name,
    location,
    experience_years
FROM Candidates;

SELECT *
FROM Candidate_Public_View;

CREATE VIEW Job_Public_View AS
SELECT
    job_id,
    job_title,
    company_id,
    location,
    employment_type
FROM Jobs;

SELECT *
FROM Job_Public_View;
